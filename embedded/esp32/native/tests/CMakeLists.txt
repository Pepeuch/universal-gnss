cmake_minimum_required(VERSION 3.22)
project(ug_embedded_native_tests LANGUAGES C CXX)
get_filename_component(UG_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(BUILD_TESTING OFF)
foreach(component core/gnss_core protocols/gnss_protocols core/gnss_transport core/gnss_driver core/gnss_ntrip)
    get_filename_component(name "${component}" NAME)
    add_subdirectory("${UG_ROOT}/${component}" "${name}")
endforeach()
add_library(ug_embedded_gateway ../../components/ug_gateway/src/ug_gateway.cpp
    ../../components/ug_rtcm/src/rtcm.cpp)
target_include_directories(ug_embedded_gateway PUBLIC ../../components/ug_gateway/include
    ../../components/ug_rtcm/include)
target_link_libraries(ug_embedded_gateway PUBLIC universal_gnss_protocols)
target_compile_features(ug_embedded_gateway PUBLIC cxx_std_17)
add_executable(test_gateway test_gateway.cpp)
target_compile_options(test_gateway PRIVATE -UNDEBUG)
target_link_libraries(test_gateway PRIVATE ug_embedded_gateway)
enable_testing()
add_test(NAME embedded_gateway_contract COMMAND test_gateway)

add_library(ug_embedded_receiver ../../components/ug_receiver_runtime/src/ug_receiver_runtime.cpp)
target_include_directories(ug_embedded_receiver PUBLIC ../../components/ug_receiver_runtime/include)
target_link_libraries(ug_embedded_receiver PUBLIC universal_gnss_driver)
add_executable(test_receiver test_receiver.cpp)
target_compile_options(test_receiver PRIVATE -UNDEBUG)
target_link_libraries(test_receiver PRIVATE ug_embedded_receiver)
add_test(NAME embedded_receiver_contract COMMAND test_receiver)
add_executable(test_base_profile test_base_profile.cpp)
target_compile_options(test_base_profile PRIVATE -UNDEBUG)
target_link_libraries(test_base_profile PRIVATE universal_gnss_driver)
add_test(NAME embedded_base_profile_contract COMMAND test_base_profile)
