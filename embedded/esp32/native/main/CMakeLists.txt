idf_build_get_property(UG_PROFILE UG_PROFILE)
idf_build_get_property(UG_LOCAL_GNSS UG_LOCAL_GNSS)
set(deps espnow_rtcm esp_driver_uart nvs_flash esp_netif esp_event esp_timer)
if(UG_PROFILE STREQUAL "rover")
    set(UG_ROVER_UART_TX -1 CACHE STRING "Explicit C3 rover TX-only GPIO, -1 for USB")
    set(UG_ROVER_UART_BAUD 115200 CACHE STRING "C3 rover UART 8N1 baud")
    set(rover_sources "rover.c")
    if(NOT UG_ROVER_UART_TX STREQUAL "-1")
        if(NOT CONFIG_IDF_TARGET_ESP32C3 OR NOT UG_ROVER_UART_TX MATCHES "^(4|5|6|7|10)$" OR
                NOT UG_ROVER_UART_BAUD MATCHES "^(115200|230400|460800|921600)$")
            message(FATAL_ERROR "UART rover requires explicit reviewed C3 TX GPIO and supported baud")
        endif()
        list(APPEND rover_sources "rover_uart.c")
    endif()
    idf_component_register(SRCS ${rover_sources} REQUIRES ${deps})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE UG_ROVER_UART_TX=${UG_ROVER_UART_TX}
        UG_ROVER_UART_BAUD=${UG_ROVER_UART_BAUD})
else()
    list(APPEND deps ug_gateway_service ug_network ug_ethernet ug_platform ug_services ug_web esp_eth)
    if(UG_LOCAL_GNSS)
        list(APPEND deps ug_gnss)
    endif()
    idf_component_register(SRCS "gateway.cpp" REQUIRES ${deps})
    target_compile_definitions(${COMPONENT_LIB} PRIVATE UG_PROFILE_NAME="${UG_PROFILE}")
    if(UG_LOCAL_GNSS)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE UG_LOCAL_GNSS=1)
    else()
        target_compile_definitions(${COMPONENT_LIB} PRIVATE UG_LOCAL_GNSS=0)
    endif()
endif()
