idf_build_get_property(UG_PROFILE UG_PROFILE)
idf_build_get_property(UG_LOCAL_GNSS UG_LOCAL_GNSS)
set(deps espnow_rtcm esp_driver_uart nvs_flash esp_netif esp_event esp_timer)
if(UG_PROFILE STREQUAL "rover")
    idf_component_register(SRCS "rover.c" REQUIRES ${deps})
else()
    list(APPEND deps ug_gateway)
    if(UG_LOCAL_GNSS)
        list(APPEND deps ug_receiver_runtime)
    endif()
    idf_component_register(SRCS "gateway.cpp" REQUIRES ${deps})
    if(UG_LOCAL_GNSS)
        target_compile_definitions(${COMPONENT_LIB} PRIVATE UG_LOCAL_GNSS=1)
    else()
        target_compile_definitions(${COMPONENT_LIB} PRIVATE UG_LOCAL_GNSS=0)
    endif()
endif()
